# ---- HMAC (currently NOT required - see AUTH_FINDINGS.md) ----
# The Postman collection sends x-hmac-meta/x-hmac-signature headers, but
# verification during suite setup showed the gateway accepts bearer-only
# auth on every service tested (finex, insurance, payout - reads and
# mutation validation). HMAC signing is implemented but disabled by
# default. Only set this if a specific endpoint starts requiring it.
# HMAC_SECRET=
SEND_HMAC_HEADERS=0

# ---- HTTP behavior ----
HTTP_TIMEOUT=20
HTTP_RETRIES=2
HTTP_DEBUG=0

# ---- Safety gate for mutation tests ----
# Mutations (create_loan, save_*, update_lead_status, UploadDocument, ...)
# are skipped unless this is set to 1, AND the active environment profile
# allows mutations (prod never does, regardless of this flag), AND the
# resolved BASE_URL host is in MUTATION_SAFE_HOSTS (src/config/environments.py)
# - this last check exists so pointing BASE_URL at prod can't silently run
# mutations there even with ENV=pre and ALLOW_MUTATIONS=1.
#
# Each mutation additionally needs an entry in
# src/testdata/mutation_profiles.py to run at all (default-deny) - see
# that file for what data gets substituted and why almost none of these
# can be cleaned up afterwards (the captured spec has no delete/revert
# mutations, so every run leaves a permanent record on `pre`).
ALLOW_MUTATIONS=0

# Second opt-in, required IN ADDITION to ALLOW_MUTATIONS=1, for the
# tier="external" mutations that reach a real lender or credit bureau
# (India Shelter lead creation, a live CIBIL bureau pull) and are
# categorically irreversible. Leave this off unless you specifically
# intend to hit those.
ALLOW_EXTERNAL_MUTATIONS=0
